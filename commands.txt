# Create key pair
aws ec2 create-key-pair \
  --key-name my-aws-keypair \
  --query "KeyMaterial" \
  --output text > my-aws-keypair.pem


# Create OIDC provider
aws iam create-open-id-connect-provider \
  --url "https://token.actions.githubusercontent.com" \
  --client-id-list "sts.amazonaws.com" \
  --thumbprint-list "6938fd4d98bab03faadb97b34396831e3780aea1"

# Create S3 Bucket
aws s3api create-bucket \
  --bucket thinhdan905 \
  --region us-east-1

# Create DynamoDB table
aws dynamodb create-table \
  --table-name terraform-lock-table \
  --attribute-definitions AttributeName=LockID,AttributeType=S \
  --key-schema AttributeName=LockID,KeyType=HASH \
  --billing-mode PAY_PER_REQUEST

# Check file .tfstate (after apply terraform)
aws s3 ls s3://thinhdan905/project/infra/

# Configure S3 bucket versioning and encryption
aws s3api put-bucket-versioning \
  --bucket thinhdan905 \
  --versioning-configuration Status=Enabled

cd aws\s3api
aws s3api put-bucket-encryption \
  --bucket thinhdan905 \
  --server-side-encryption-configuration file://encryption.json

# Create IAM Role for GitHub Actions
cd aws\oidc
aws iam create-role \
  --role-name GitHubActions-TerraformRole \
  --assume-role-policy-document file://trust-policy.json

# Create IAM Policy for GitHub Actions
aws iam create-policy --policy-name GitHubActions-TerraformPolicy --policy-document file://tf-policy.json

# Attach IAM Policy to IAM Role (remember to replace ACCOUNT_ID with your actual AWS account ID)
aws iam attach-role-policy --role-name GitHubActions-TerraformRole --policy-arn arn:aws:iam::ACCOUNT_ID:policy/GitHubActions-TerraformPolicy

# Check IAM Role
aws iam get-role --role-name GitHubActions-TerraformRole

# Check attached policies to the IAM Role
aws iam list-attached-role-policies --role-name GitHubActions-TerraformRole
